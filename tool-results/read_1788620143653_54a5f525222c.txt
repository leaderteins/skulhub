    93→    93→function Sparkline({ data, color }: { data: number[]; color: string }) {
    94→    94→  if (!data.length) return <div className="h-10 w-full" />
    95→    95→  const chartData = data.map((v, i) => ({ i, v }))
    96→    96→  const id = `notif-spark-${color.replace('#', '')}-${Math.random().toString(36).slice(2, 8)}`
    97→    97→  return (
    98→    98→    <div className="h-10 w-full">
    99→    99→      <ResponsiveContainer width="100%" height="100%">
   100→   100→        <AreaChart data={chartData} margin={{ top: 2, right: 0, left: 0, bottom: 0 }}>
   101→   101→          <defs>
   102→   102→            <linearGradient id={id} x1="0" y1="0" x2="0" y2="1">
   103→   103→              <stop offset="0%" stopColor={color} stopOpacity={0.35} />
   104→   104→              <stop offset="100%" stopColor={color} stopOpacity={0} />
   105→   105→            </linearGradient>
   106→   106→          </defs>
   107→   107→          <Area
   108→   108→            type="monotone"
   109→   109→            dataKey="v"
   110→   110→            stroke={color}
   111→   111→            strokeWidth={2}
   112→   112→            fill={`url(#${id})`}
   113→   113→            isAnimationActive={false}
   114→   114→            dot={false}
   115→   115→          />
   116→   116→        </AreaChart>
   117→   117→      </ResponsiveContainer>

   122→   122→function DeltaPill({ delta, label }: { delta: number | null; label?: string }) {
   123→   123→  if (delta === null || !isFinite(delta)) return null
   124→   124→  const up = delta >= 0
   125→   125→  return (
   126→   126→    <span className={cn(
   127→   127→      'inline-flex items-center gap-0.5 rounded-full px-1.5 py-0.5 text-[10px] font-semibold',
   128→   128→      up ? 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400'
   129→   129→        : 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-400',

   236→   236→export function NotificationsModule() {
   237→   237→  const { data, loading, refetch } = useFetch<NotificationsResponse>('/api/notifications?limit=200')
   238→   238→
   239→   239→  const [search, setSearch] = useState('')

   245→   245→  const [seeded, setSeeded] = useState(false)
   246→   246→
   247→   247→  // Listen for the realtime 'notifications:refetch' custom event (dispatched
   248→   248→  // by the useRealtimeNotifications hook) and refetch when it fires.
   249→   249→  useEffect(() => {
   250→   250→    const handler = () => refetch()
   251→   251→    window.addEventListener('notifications:refetch', handler)
   252→   252→    return () => window.removeEventListener('notifications:refetch', handler)
   253→   253→  }, [refetch])
   254→   254→
   255→   255→  // When the user is signed in, auto-seed demo data the first time so the
   256→   256→  // redesigned UI has something to show. We only do this if there's no data
   257→   257→  // AND we've never seeded before in this session.
   258→   258→  useEffect(() => {
   259→   259→    if (loading || seeded) return
   260→   260→    if (data && data.notifications.length === 0 && (data.stats?.total ?? 0) === 0) {
   261→   261→      setSeeded(true)
   262→   262→      ;(async () => {
   263→   263→        try {
   264→   264→          const res = await fetch('/api/notifications/seed', {
   265→   265→            method: 'POST',
   266→   266→            headers: { 'Content-Type': 'application/json' },
   267→   267→            body: JSON.stringify({}),
   268→   268→          })
   269→   269→          if (res.ok) refetch()
   270→   270→        } catch {}
   271→   271→      })()
   272→   272→    }
   273→   273→  }, [data, loading, refetch, seeded])
   274→   274→
   275→   275→  const markAllRead = useCallback(async () => {
   276→   276→    setBusy(true)
   277→   277→    try {
   278→   278→      const res = await fetch('/api/notifications/read-all', {
   279→   279→        method: 'POST',
   280→   280→        headers: { 'Content-Type': 'application/json' },
   281→   281→        body: JSON.stringify({ scope: 'school' }),
   282→   282→      })
   283→   283→      if (!res.ok) throw new Error('Failed to mark all read')
   284→   284→      toast.success('All notifications marked as read')
   285→   285→      refetch()
   286→   286→    } catch (e: unknown) {
   287→   287→      toast.error(e instanceof Error ? e.message : 'Operation failed')
   288→   288→    } finally {
   289→   289→      setBusy(false)
   290→   290→    }
   291→   291→  }, [refetch])
   292→   292→
   293→   293→  const clearAll = useCallback(async () => {
   294→   294→    setBusy(true)
   295→   295→    try {
   296→   296→      const res = await fetch('/api/notifications/seed', { method: 'DELETE' })
   297→   297→      if (!res.ok) throw new Error('Failed to clear notifications')
   298→   298→      toast.success('All notifications cleared')
   299→   299→      setClearOpen(false)
   300→   300→      refetch()
   301→   301→    } catch (e: unknown) {
   302→   302→      toast.error(e instanceof Error ? e.message : 'Operation failed')
   303→   303→    } finally {
   304→   304→      setBusy(false)
   305→   305→    }
   306→   306→  }, [refetch])
   307→   307→
   308→   308→  const markOneRead = useCallback(async (n: AppNotificationDTO) => {
   309→   309→    try {
   310→   310→      await fetch(`/api/notifications/${n.id}`, {
   311→   311→        method: 'PATCH',

   324→   324→        method: 'PATCH',
   325→   325→        headers: { 'Content-Type': 'application/json' },

   330→   330→    } catch {
   331→   331→      toast.error('Could not archive')
   332→   332→    }
   333→   333→  }, [refetch])
   334→   334→
   335→   335→  const deleteOne = useCallback(async (n: AppNotificationDTO) => {
   336→   336→    try {

   338→   338→      toast.success(`Deleted: ${n.title}`)

   340→   340→    } catch {
   341→   341→      toast.error('Could not delete')
   342→   342→    }

   344→   344→

   356→   356→    } catch (e: unknown) {
   357→   357→      toast.error(e instanceof Error ? e.message : 'Reseed failed')
   358→   358→    } finally {
   359→   359→      setBusy(false)
   360→   360→    }
   361→   361→  }, [refetch])
   362→   362→
   363→   363→  // ---- Loading skeleton mirrors final layout ----
   364→   364→  if (loading || !data) {
   365→   365→    return (
   366→   366→      <div className="space-y-6">
   367→   367→        <Skeleton className="h-44 w-full rounded-2xl" />
   368→   368→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
   369→   369→          {Array.from({ length: 4 }).map((_, i) => (
   370→   370→            <Skeleton key={i} className="h-32 rounded-xl" />
   371→   371→          ))}
   372→   372→        </div>
   373→   373→        <div className="grid gap-4 lg:grid-cols-3">
   374→   374→          <Skeleton className="h-80 rounded-xl lg:col-span-1" />
   375→   375→          <Skeleton className="h-80 rounded-xl lg:col-span-2" />
   376→   376→        </div>
   377→   377→        <Skeleton className="h-96 rounded-xl" />
   378→   378→      </div>
   379→   379→    )
   380→   380→  }
   381→   381→
   382→   382→  // ---- Safe defaults ----
   383→   383→  const notifications = data.notifications ?? []
   384→   384→  const stats = data.stats ?? { total: 0, unread: 0, actionRequired: 0, today: 0, byType: {} }
   385→   385→  const byDay = data.byDay ?? []
   386→   386→  const byType = stats.byType ?? {}
   387→   387→  const total = stats.total ?? 0
   388→   388→  const unread = stats.unread ?? 0
   389→   389→  const actionRequired = stats.actionRequired ?? 0
   390→   390→  const today = stats.today ?? 0
   391→   391→
   392→   392→  // Derive sparkline data from byDay (last 7 days) and type distributions
   393→   393→  const dailySpark = byDay.map((d) => d.count || 0)
   394→   394→  const totalDelta = dailySpark.length >= 2 ? ((dailySpark[dailySpark.length - 1] - dailySpark[dailySpark.length - 2]) / Math.max(1, dailySpark[dailySpark.length - 2])) * 100 : null
   395→   395→
   396→   396→  // Build donut data
   397→   397→  const donutData = Object.keys(TYPE_META).map((k) => ({
   398→   398→    type: k,
   399→   399→    label: TYPE_META[k].label,
   400→   400→    value: byType[k] || 0,
   401→   401→    color: TYPE_META[k].color,
   402→   402→  })).filter((d) => d.value > 0)
   403→   403→
   404→   404→  // Filter the notifications for the table
   405→   405→  const filtered = notifications.filter((n) => {
   406→   406→    if (search) {
   407→   407→      const q = search.toLowerCase()
   408→   408→      if (!n.title.toLowerCase().includes(q) && !n.message.toLowerCase().includes(q)) return false
   409→   409→    }
   410→   410→    if (typeFilter !== 'all' && n.type !== typeFilter) return false
   411→   411→    if (priorityFilter !== 'all' && n.priority !== priorityFilter) return false
