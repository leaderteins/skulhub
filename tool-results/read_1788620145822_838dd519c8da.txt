    93→function Sparkline({ data, color }: { data: number[]; color: string }) {
    94→  if (!data.length) return <div className="h-10 w-full" />
    95→  const chartData = data.map((v, i) => ({ i, v }))
    96→  const id = `notif-spark-${color.replace('#', '')}-${Math.random().toString(36).slice(2, 8)}`
    97→  return (
    98→    <div className="h-10 w-full">
    99→      <ResponsiveContainer width="100%" height="100%">
   100→        <AreaChart data={chartData} margin={{ top: 2, right: 0, left: 0, bottom: 0 }}>
   101→          <defs>
   102→            <linearGradient id={id} x1="0" y1="0" x2="0" y2="1">
   103→              <stop offset="0%" stopColor={color} stopOpacity={0.35} />
   104→              <stop offset="100%" stopColor={color} stopOpacity={0} />
   105→            </linearGradient>
   106→          </defs>
   107→          <Area
   108→            type="monotone"
   109→            dataKey="v"
   110→            stroke={color}
   111→            strokeWidth={2}
   112→            fill={`url(#${id})`}
   113→            isAnimationActive={false}
   114→            dot={false}
   115→          />
   116→        </AreaChart>
   117→      </ResponsiveContainer>
   118→    </div>
   119→  )
   120→}

   122→function DeltaPill({ delta, label }: { delta: number | null; label?: string }) {
   123→  if (delta === null || !isFinite(delta)) return null
   124→  const up = delta >= 0
   125→  return (
   126→    <span className={cn(
   127→      'inline-flex items-center gap-0.5 rounded-full px-1.5 py-0.5 text-[10px] font-semibold',
   128→      up ? 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400'
   129→        : 'bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-400',
   130→    )}>
   131→      {up ? <ChevronUp className="h-3 w-3" /> : <ChevronDown className="h-3 w-3" />}
   132→      {Math.abs(delta).toFixed(0)}
   133→      {label && <span className="ml-1 font-normal opacity-70">{label}</span>}
   134→    </span>
   135→  )
   136→}

   258→  useEffect(() => {

   261→      setSeeded(true)
   262→      ;(async () => {
   263→        try {
   264→          const res = await fetch('/api/notifications/seed', {
   265→            method: 'POST',
   266→            headers: { 'Content-Type': 'application/json' },
   267→            body: JSON.stringify({}),
   268→          })
   269→          if (res.ok) refetch()
   270→        } catch {}
   271→      })()
   272→    }
   273→  }, [data, loading, refetch, seeded])
   274→

   278→      const res = await fetch('/api/notifications/read-all', {
   279→        method: 'POST',
   280→        headers: { 'Content-Type': 'application/json' },
   281→        body: JSON.stringify({ scope: 'school' }),
   282→      })
   283→      if (!res.ok) throw new Error('Failed to mark all read')
   284→      toast.success('All notifications marked as read')
   285→      refetch()
   286→    } catch (e: unknown) {
   287→      toast.error(e instanceof Error ? e.message : 'Operation failed')
   288→    } finally {
   289→      setBusy(false)
   290→    }
   291→  }, [refetch])
   292→
   293→  const clearAll = useCallback(async () => {
   294→    setBusy(true)

   297→      if (!res.ok) throw new Error('Failed to clear notifications')
   298→      toast.success('All notifications cleared')
   299→      setClearOpen(false)
   300→      refetch()
   301→    } catch (e: unknown) {
   302→      toast.error(e instanceof Error ? e.message : 'Operation failed')
   303→    } finally {
   304→      setBusy(false)
   305→    }
   306→  }, [refetch])
   307→
   308→  const markOneRead = useCallback(async (n: AppNotificationDTO) => {
   309→    try {
   310→      await fetch(`/api/notifications/${n.id}`, {
   311→        method: 'PATCH',
   312→        headers: { 'Content-Type': 'application/json' },
   313→        body: JSON.stringify({ read: true }),
   314→      })
   315→      refetch()
   316→    } catch {
   317→      toast.error('Could not mark as read')
   318→    }
   319→  }, [refetch])
   320→
   321→  const archiveOne = useCallback(async (n: AppNotificationDTO) => {
   322→    try {
   323→      await fetch(`/api/notifications/${n.id}`, {
   324→        method: 'PATCH',
   325→        headers: { 'Content-Type': 'application/json' },
   326→        body: JSON.stringify({ archived: true }),
   327→      })

   329→      refetch()
   330→    } catch {
   331→      toast.error('Could not archive')

   335→  const deleteOne = useCallback(async (n: AppNotificationDTO) => {

   343→  }, [refetch])

   349→        method: 'POST',

   386→  const byType = stats.byType ?? {}

   394→  const totalDelta = dailySpark.length >= 2 ? ((dailySpark[dailySpark.length - 1] - dailySpark[dailySpark.length - 2]) / Math.max(1, dailySpark[dailySpark.length - 2])) * 100 : null
   395→
