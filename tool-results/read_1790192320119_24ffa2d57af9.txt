   159→
   160→export function ParentPortal() {
   161→  const { setAuthView } = useAuthStore()
   162→  const [schoolCode, setSchoolCode] = useState('')
   163→  const [admissionNo, setAdmissionNo] = useState('')
   164→  const [phone, setPhone] = useState('')
   165→  const [loading, setLoading] = useState(false)
   166→  const [studentId, setStudentId] = useState<string | null>(null)
   167→  const [schoolName, setSchoolName] = useState<string>('')
   168→  const [paying, setPaying] = useState<string | null>(null)
   169→  const [payDialog, setPayDialog] = useState<{ invoiceId: string; amount: number } | null>(null)
   170→  const [payPhone, setPayPhone] = useState('')
   171→
   172→  const {
   173→    data: dashboard,
   174→    loading: dashLoading,
   175→    error,
   176→    refetch,
   177→  } = useFetch<DashboardData>(studentId ? `/api/parent/${studentId}` : null)
   178→
   179→  // --- Step 1: verify identity ------------------------------------------
   180→  const handleLookup = async (e: React.FormEvent) => {
   181→    e.preventDefault()
   182→    if (!schoolCode.trim() || !admissionNo.trim() || !phone.trim()) {
   183→      toast.error('All fields are required')
   184→      return
   185→    }
   186→    setLoading(true)
   187→    try {
   188→      const res = await fetch('/api/parent/lookup', {
   189→        method: 'POST',
   190→        headers: { 'Content-Type': 'application/json' },
   191→        body: JSON.stringify({
   192→          schoolCode: schoolCode.trim(),
   193→          admissionNo: admissionNo.trim(),
   194→          phone: phone.trim(),
   195→        }),
   196→      })
   197→      const data = await res.json().catch(() => ({}))
   198→      if (!res.ok) {
   199→        toast.error('Lookup failed', {
   200→          description: data?.error || 'Please try again.',
   201→        })
   202→        setLoading(false)
   203→        return
   204→      }
   205→      const result = data as LookupResult
   206→      toast.success('Verified', {
   207→        description: `${result.student.name} · ${result.schoolName}`,
   208→      })
   209→      setStudentId(result.student.id)
   210→      setSchoolName(result.schoolName)
   211→    } catch {
   212→      toast.error('Lookup failed', { description: 'Please try again.' })
   213→    }
   214→    setLoading(false)
   215→  }
   216→
   217→  const fillDemo = async () => {
   218→    try {
   219→      const res = await fetch('/api/parent/demo')
   220→      const data = await res.json().catch(() => ({}))
   221→      if (!res.ok) {
   222→        toast.error('No demo data available', {
   223→          description: data?.error || 'Seed the database first.',
   224→        })
   225→        return
   226→      }
   227→      setSchoolCode(data.schoolCode)
   228→      setAdmissionNo(data.admissionNo)
   229→      setPhone(data.phone)
   230→      toast.success('Demo credentials filled', {
   231→        description: `${data.studentName} at ${data.schoolName}`,
   232→      })
   233→    } catch {
   234→      toast.error('Could not load demo credentials')
   235→    }
   236→  }
   237→
   238→  const exitPortal = () => {
   239→    setStudentId(null)
   240→    setSchoolCode('')
   241→    setAdmissionNo('')
   242→    setPhone('')
   243→    setSchoolName('')
   244→  }
   245→
   246→  // --- M-Pesa payment handler ---
   247→  const handlePayInvoice = (invoiceId: string) => {
   248→    const inv = dashboard?.fees?.invoices?.find(i => i.id === invoiceId)
   249→    if (!inv) return
   250→    setPayDialog({ invoiceId, amount: inv.balance })
   251→    setPayPhone(dashboard?.guardian?.phone || phone || '')
   252→  }
   253→
   254→  const confirmPayment = async () => {
   255→    if (!payDialog || !studentId) return
   256→    if (!payPhone.trim()) { toast.error('Phone number is required'); return }
   257→    setPaying(payDialog.invoiceId)
   258→    try {
   259→      const res = await fetch('/api/mpesa/parent-pay', {
   260→        method: 'POST',
   261→        headers: { 'Content-Type': 'application/json' },
   262→        body: JSON.stringify({
   263→          studentId,
   264→          invoiceId: payDialog.invoiceId,
   265→          phone: payPhone.trim(),
   266→        }),
   267→      })
   268→      const data = await res.json()
   269→      if (!res.ok) throw new Error(data.error)
   270→
   271→      if (data.demo) {
   272→        toast.success('Payment logged (demo mode)', {
   273→          description: data.message,
   274→          duration: 6000,
   275→        })
   276→      } else {
   277→        toast.success('STK Push sent!', {
   278→          description: `Check your phone (${data.phone}) and enter your M-Pesa PIN to pay KES ${data.amount?.toLocaleString()}.`,
   279→          duration: 10000,
   280→        })
   281→      }
   282→      setPayDialog(null)
   283→      setTimeout(() => refetch(), 3000) // refresh after 3s
   284→    } catch (e: any) {
   285→      toast.error('Payment failed', { description: e.message })
   286→    } finally {
   287→      setPaying(null)
   288→    }
   289→  }
   290→
   291→  // --- Dashboard view ---------------------------------------------------
   292→  if (studentId) {
   293→    return (
   294→      <div className="min-h-screen bg-gradient-to-br from-emerald-50 via-teal-50 to-cyan-50 dark:from-slate-950 dark:via-slate-900 dark:to-emerald-950">
   295→        <header className="sticky top-0 z-10 border-b bg-white/80 backdrop-blur dark:bg-slate-900/80">
   296→          <div className="mx-auto flex max-w-6xl flex-wrap items-center justify-between gap-3 px-4 py-3">
   297→            <div className="flex items-center gap-3">
   298→              <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-gradient-to-br from-emerald-500 to-teal-600 text-white">
   299→                <School className="h-5 w-5" />
   300→              </div>
   301→              <div>
   302→                <p className="text-sm font-bold text-foreground">Parent Portal</p>
   303→                <p className="text-xs text-muted-foreground">
   304→                  {schoolName || dashboard?.school?.name || 'SkulHub'}
   305→                </p>
   306→              </div>
   307→            </div>
   308→            <div className="flex flex-wrap items-center gap-2">
   309→              <Button
   310→                variant="outline"
   311→                size="sm"
   312→                onClick={refetch}
   313→                disabled={dashLoading}
   314→              >
   315→                <RefreshCw
   316→                  className={`mr-2 h-3.5 w-3.5 ${dashLoading ? 'animate-spin' : ''}`}
   317→                />
   318→                Refresh
   319→              </Button>
   320→              <Button
   321→                variant="outline"
   322→                size="sm"
   323→                onClick={() => setAuthView('login')}
   324→              >
   325→                <ArrowLeft className="mr-2 h-3.5 w-3.5" /> To login
   326→              </Button>
   327→              <Button variant="ghost" size="sm" onClick={exitPortal}>
   328→                <LogOut className="mr-2 h-3.5 w-3.5" /> Exit
   329→              </Button>
   330→            </div>
   331→          </div>
   332→        </header>
   333→
   334→        <main className="mx-auto max-w-6xl px-4 py-6">
   335→          {dashLoading && !dashboard && (
   336→            <div className="flex items-center justify-center py-20">
   337→              <div className="h-10 w-10 animate-spin rounded-full border-4 border-emerald-500/20 border-t-emerald-600" />
   338→            </div>
   339→          )}
   340→
   341→          {error && !dashLoading && (
   342→            <Card>
   343→              <CardContent className="flex flex-col items-center justify-center gap-3 py-12 text-center">
   344→                <AlertCircle className="h-8 w-8 text-rose-500" />
   345→                <p className="text-sm font-semibold">Failed to load dashboard</p>
   346→                <p className="text-xs text-muted-foreground">{error}</p>
   347→                <Button size="sm" variant="outline" onClick={refetch}>
   348→                  Retry
   349→                </Button>
   350→              </CardContent>
   351→            </Card>
   352→          )}
   353→
   354→          {dashboard && (
   355→            <div className="space-y-6">
   356→              {/* Student header */}
   357→              <Card className="border-0 shadow-md">
   358→                <CardContent className="flex flex-col gap-4 p-5 sm:flex-row sm:items-center">
   359→                  <Avatar className="h-16 w-16 border-2 border-emerald-200">
   360→                    <AvatarFallback className="bg-gradient-to-br from-emerald-500 to-teal-600 text-lg font-bold text-white">
   361→                      {initials(dashboard.student?.firstName, dashboard.student?.lastName)}
   362→                    </AvatarFallback>
   363→                  </Avatar>
   364→                  <div className="flex-1">
   365→                    <div className="flex flex-wrap items-center gap-2">
   366→                      <h2 className="text-xl font-bold text-foreground">
   367→                        {dashboard.student?.name || 'Student'}
   368→                      </h2>
   369→                      <Badge variant="outline" className={statusColor(dashboard.student?.status || 'Active')}>
   370→                        {dashboard.student?.status || 'Active'}
   371→                      </Badge>
   372→                      <Badge
   373→                        variant="outline"
   374→                        className={
   375→                          dashboard.student?.boarding
   376→                            ? 'bg-violet-100 text-violet-700 dark:bg-violet-950 dark:text-violet-400'
   377→                            : 'bg-cyan-100 text-cyan-700 dark:bg-cyan-950 dark:text-cyan-400'
   378→                        }
   379→                      >
   380→                        {dashboard.student?.boarding ? 'Boarding' : 'Day Scholar'}
   381→                      </Badge>
   382→                    </div>
   383→                    <p className="mt-1 text-sm text-muted-foreground">
   384→                      Admission No:{' '}
   385→                      <span className="font-medium text-foreground">
   386→                        {dashboard.student?.admissionNo || '—'}
   387→                      </span>
   388→                      {' · '}Class:{' '}
   389→                      <span className="font-medium text-foreground">
   390→                        {dashboard.student?.classLevel || '—'}
   391→                      </span>
   392→                      {' · '}Stream:{' '}
   393→                      <span className="font-medium text-foreground">
   394→                        {dashboard.student?.stream || '—'}
   395→                      </span>
   396→                    </p>
   397→                    {dashboard.guardian && (
   398→                      <p className="mt-1 text-xs text-muted-foreground">
   399→                        Guardian: {dashboard.guardian.name} ({dashboard.guardian.relation}) ·{' '}
   400→                        {dashboard.guardian.phone}
   401→                      </p>
   402→                    )}
   403→                  </div>
   404→                </CardContent>
   405→              </Card>
   406→
   407→              <div className="grid gap-6 lg:grid-cols-3">
   408→                {/* Fee summary */}
   409→                <Card className="lg:col-span-2">
   410→                  <CardHeader>
   411→                    <CardTitle className="flex items-center gap-2 text-base">
   412→                      <Wallet className="h-4 w-4 text-emerald-600" /> Fee Summary
   413→                    </CardTitle>
   414→                  </CardHeader>
   415→                  <CardContent>
   416→                    <div className="grid grid-cols-3 gap-3">
   417→                      <div className="rounded-xl border bg-slate-50/50 p-3 dark:bg-slate-900/30">
   418→                        <p className="text-xs text-muted-foreground">Total Billed</p>
   419→                        <p className="text-base font-bold text-foreground sm:text-lg">
   420→                          {formatKES(dashboard.fees.totalBilled)}
   421→                        </p>
   422→                      </div>
   423→                      <div className="rounded-xl border bg-emerald-50/50 p-3 dark:bg-emerald-950/20">
   424→                        <p className="text-xs text-muted-foreground">Paid</p>
   425→                        <p className="text-base font-bold text-emerald-700 dark:text-emerald-400 sm:text-lg">
   426→                          {formatKES(dashboard.fees.totalPaid)}
   427→                        </p>
   428→                      </div>
   429→                      <div className="rounded-xl border bg-rose-50/50 p-3 dark:bg-rose-950/20">
   430→                        <p className="text-xs text-muted-foreground">Balance</p>
   431→                        <p className="text-base font-bold text-rose-700 dark:text-rose-400 sm:text-lg">
   432→                          {formatKES(dashboard.fees.totalBalance)}
   433→                        </p>
   434→                      </div>
   435→                    </div>
   436→                    {dashboard.fees.invoices.length > 0 ? (
   437→                      <div className="mt-4 max-h-64 overflow-y-auto pr-1">
   438→                        <table className="w-full text-sm">
   439→                          <thead>
   440→                            <tr className="border-b text-xs text-muted-foreground">
   441→                              <th className="py-2 text-left font-medium">Invoice</th>
   442→                              <th className="py-2 text-left font-medium">Term</th>
   443→                              <th className="py-2 text-right font-medium">Amount</th>
   444→                              <th className="py-2 text-right font-medium">Balance</th>
   445→                              <th className="py-2 text-left font-medium">Status</th>
   446→                              <th className="py-2 text-right font-medium">Action</th>
   447→                            </tr>
   448→                          </thead>
   449→                          <tbody>
   450→                            {dashboard.fees.invoices.map(inv => (
   451→                              <tr key={inv.id} className="border-b last:border-0">
   452→                                <td className="py-2 font-mono text-xs">{inv.invoiceNo}</td>
   453→                                <td className="py-2 text-xs">
   454→                                  {inv.term} {inv.academicYear}
   455→                                </td>
   456→                                <td className="py-2 text-right">{formatKES(inv.amount)}</td>
   457→                                <td className="py-2 text-right">{formatKES(inv.balance)}</td>
   458→                                <td className="py-2">
   459→                                  <span
   460→                                    className={`rounded px-2 py-0.5 text-xs ${statusColor(inv.status)}`}
   461→                                  >
   462→                                    {inv.status}
   463→                                  </span>
   464→                                </td>
   465→                                <td className="py-2 text-right">
   466→                                  {inv.balance > 0 && inv.status !== 'Paid' && inv.status !== 'Cancelled' && (
   467→                                    <Button
   468→                                      size="sm"
   469→                                      className="bg-emerald-600 hover:bg-emerald-700 text-xs h-7"
   470→                                      onClick={() => handlePayInvoice(inv.id)}
   471→                                      disabled={paying === inv.id}
   472→                                    >
   473→                                      {paying === inv.id ? (
   474→                                        <><Loader2 className="mr-1 h-3 w-3 animate-spin" /> Sending...</>
   475→                                      ) : (
   476→                                        <><Wallet className="mr-1 h-3 w-3" /> Pay M-Pesa</>
   477→                                      )}
   478→                                    </Button>
   479→                                  )}
   480→                                </td>
   481→                              </tr>
   482→                            ))}
   483→                          </tbody>
   484→                        </table>
   485→                      </div>
   486→                    ) : (
   487→                      <p className="mt-4 text-sm text-muted-foreground">No invoices yet.</p>
   488→                    )}
   489→                  </CardContent>
   490→                </Card>
   491→
   492→                {/* Recent attendance */}
   493→                <Card>
   494→                  <CardHeader>
   495→                    <CardTitle className="flex items-center gap-2 text-base">
   496→                      <CalendarDays className="h-4 w-4 text-teal-600" /> Recent Attendance
   497→                    </CardTitle>
   498→                  </CardHeader>
   499→                  <CardContent>
   500→                    {dashboard.attendance.length === 0 ? (
   501→                      <p className="text-sm text-muted-foreground">No records yet.</p>
   502→                    ) : (
   503→                      <div className="max-h-72 space-y-2 overflow-y-auto pr-1">
   504→                        {dashboard.attendance.map(a => (
   505→                          <div
   506→                            key={a.id}
   507→                            className="flex items-center justify-between gap-2 rounded-lg border p-2"
   508→                          >
   509→                            <span className="text-sm text-muted-foreground">
   510→                              {formatDate(a.date, { weekday: 'short', month: 'short', day: 'numeric' })}
   511→                            </span>
   512→                            <span
   513→                              className={`rounded px-2 py-0.5 text-xs font-medium ${
   514→                                ATTENDANCE_BADGE[a.status] ||
   515→                                'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-400'
   516→                              }`}
   517→                            >
   518→                              {a.status}
   519→                            </span>
   520→                          </div>
   521→                        ))}
   522→                      </div>
   523→                    )}
   524→                  </CardContent>
   525→                </Card>
   526→              </div>
   527→
   528→              <div className="grid gap-6 lg:grid-cols-2">
   529→                {/* Recent grades with teacher remarks */}
   530→                <Card>
   531→                  <CardHeader>
   532→                    <CardTitle className="flex items-center gap-2 text-base">
   533→                      <BookOpen className="h-4 w-4 text-emerald-600" /> Recent Grades & Remarks
   534→                    </CardTitle>
   535→                  </CardHeader>
   536→                  <CardContent>
   537→                    {dashboard.grades.length === 0 ? (
   538→                      <p className="text-sm text-muted-foreground">
   539→                        No grades published yet.
   540→                      </p>
   541→                    ) : (
   542→                      <div className="space-y-3">
   543→                        {dashboard.grades.map(g => (
   544→                          <div key={g.id} className="rounded-lg border p-3">
   545→                            <div className="flex items-center justify-between gap-2">
   546→                              <div className="min-w-0">
   547→                                <p className="truncate text-sm font-medium">{g.subject}</p>
   548→                                <p className="truncate text-xs text-muted-foreground">
   549→                                  {g.exam} · {g.term}
   550→                                </p>
   551→                              </div>
   552→                              <div className="flex items-center gap-2">

   928→function ParentHomeworkSection({ admissionNo }: { admissionNo: string }) {
   929→  const [data, setData] = useState<{ homework: any[]; grades: any[] } | null>(null)
   930→  const [loading, setLoading] = useState(true)
   931→  const [expandedHw, setExpandedHw] = useState<string | null>(null)

   933→  const [posting, setPosting] = useState(false)
   934→

   936→    let cancelled = false
   937→    const raf = requestAnimationFrame(() => {
   938→      if (!admissionNo) {
   939→        setLoading(false)
   940→        return
   941→      }
   942→      fetch(`/api/parent/homework?admissionNo=${encodeURIComponent(admissionNo)}`)
   943→        .then(r => r.json())
   944→        .then(d => { if (!cancelled) { setData(d); setLoading(false) } })
   945→        .catch(() => { if (!cancelled) { setData({ homework: [], grades: [] }); setLoading(false) } })
   946→    })

   948→  }, [admissionNo])
   949→
   950→  const handlePostComment = async (hwId: string) => {
   951→    if (!commentText.trim()) return
   952→    setPosting(true)

   955→        method: 'POST',
   956→        headers: { 'Content-Type': 'application/json' },
   957→        body: JSON.stringify({ authorName: 'Guardian', authorRole: 'parent', message: commentText.trim() }),
   958→      })
   959→      if (res.ok) {
   960→        setCommentText('')
   961→        // Refresh data
   962→        fetch(`/api/parent/homework?admissionNo=${encodeURIComponent(admissionNo)}`)
   963→          .then(r => r.json())
   964→          .then(d => setData(d))
   965→      }
   966→    } catch {}
   967→    setPosting(false)
   968→  }
   969→
   970→  if (loading) {
   971→    return (
   972→      <Card>
   973→        <CardHeader>
   974→          <CardTitle className="flex items-center gap-2 text-base">
   975→            <FileText className="h-4 w-4 text-amber-600" /> Homework & Diary
   976→          </CardTitle>
   977→        </CardHeader>

   979→          <Skeleton className="h-24 w-full" />

   982→    )
   983→  }
   984→
   985→  const homework = data?.homework || []
   986→
   987→  return (
   988→    <Card>

   990→        <CardTitle className="flex items-center gap-2 text-base">

   997→        ) : (
   998→          <div className="space-y-3">
   999→            {homework.map(hw => (
  1000→              <div key={hw.id} className="rounded-lg border p-3">
  1001→                {/* Homework header */}
  1002→                <button
  1003→                  onClick={() => setExpandedHw(expandedHw === hw.id ? null : hw.id)}
  1004→                  className="flex w-full items-start justify-between gap-2 text-left"
  1005→                >
  1006→                  <div className="min-w-0">
  1007→                    <p className="text-sm font-semibold">{hw.title}</p>
  1008→                    <p className="text-xs text-muted-foreground">
  1009→                      {hw.subject?.name || 'General'} · Due {new Date(hw.dueDate).toLocaleDateString()}
  1010→                    </p>
  1011→                  </div>
  1012→                  <Badge variant="outline" className={`shrink-0 text-[9px] ${hw.status === 'Active' ? 'border-emerald-300 text-emerald-700' : 'text-muted-foreground'}`}>
  1013→                    {hw.status}
  1014→                  </Badge>
  1015→                </button>
  1016→
  1017→                {/* Expanded content */}
  1018→                {expandedHw === hw.id && (
  1019→                  <div className="mt-3 space-y-3 border-t pt-3">
  1020→                    {/* Description */}
  1021→                    <div>
  1022→                      <p className="text-[10px] font-semibold uppercase text-muted-foreground">Description</p>
  1023→                      <p className="text-sm">{hw.description}</p>
  1024→                    </div>
  1025→
  1026→                    {/* Comments diary */}
  1027→                    <div>
  1028→                      <p className="text-[10px] font-semibold uppercase text-muted-foreground">
  1029→                        Communication ({hw.comments?.length || 0})
  1030→                      </p>
  1031→                      {hw.comments && hw.comments.length > 0 ? (
