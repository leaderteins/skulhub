   414→
   415→            {/* Attendance breakdown */}
   416→            {menu.attendances.length > 0 && (
   417→              <div>
   418→                <p className="mb-2 text-sm font-semibold">Attendance Breakdown</p>
   419→                <div className="space-y-1.5">
   420→                  {menu.attendances.map((a: any) => (
   421→                    <div key={a.id} className="flex items-center justify-between rounded-lg border p-2.5">
   422→                      <div className="flex items-center gap-2">
   423→                        <Badge variant="outline" className="text-[10px]">{a.personType}</Badge>
   424→                        {a.notes && <span className="text-[10px] text-muted-foreground">{a.notes}</span>}
   425→                      </div>
   426→                      <span className="text-sm font-bold">{a.headcount} diners</span>
   427→                    </div>
   428→                  ))}
   429→                </div>
   430→              </div>
   431→            )}
   432→          </>
   433→        )}
   434→      </DialogContent>
   435→    </Dialog>
   436→  )

   442→function AddMenuDialog({ onClose }: { onClose: () => void }) {
   443→  const [form, setForm] = useState({
   444→    mealType: 'Breakfast', item: '', accompaniment: '', beverage: '', notes: '',
   445→    servingsPlanned: '200', date: '', cook: '',
   446→  })
   447→  const [saving, setSaving] = useState(false)
   448→
   449→  const handleSubmit = async () => {
   450→    if (!form.item || !form.date) { toast.error('Main dish and date are required'); return }
   451→    setSaving(true)
   452→    try {
   453→      const res = await fetch('/api/cafeteria', {
   454→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   455→        body: JSON.stringify({ ...form, status: 'Planned' }),
   456→      })
   457→      if (!res.ok) throw new Error('Failed to create menu')
   458→      toast.success('Menu created successfully')
   459→      onClose()
   460→    } catch (e: any) { toast.error(e?.message || 'Failed to create menu') }
   461→    finally { setSaving(false) }
   462→  }
   463→
   464→  return (
   465→    <Dialog open onOpenChange={(o) => { if (!o) onClose() }}>
   466→      <DialogContent className="max-w-lg max-h-[92vh] overflow-y-auto scrollbar-thin">
   467→        <DialogHeader>
   468→          <DialogTitle className="flex items-center gap-2"><Plus className="h-5 w-5 text-amber-500" /> New Meal Menu</DialogTitle>
   469→          <DialogDescription>Plan a meal for the cafeteria</DialogDescription>
   470→        </DialogHeader>
   471→        <div className="space-y-3">
   472→          <div className="grid grid-cols-2 gap-3">
   473→            <div>
   474→              <Label className="text-xs">Meal Type *</Label>
   475→              <Select value={form.mealType} onValueChange={v => setForm({ ...form, mealType: v })}>
   476→                <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   477→                <SelectContent>
   478→                  {Object.keys(MEAL_META).map(m => <SelectItem key={m} value={m}>{m}</SelectItem>)}
   479→                </SelectContent>
   480→              </Select>
   481→            </div>
   482→            <div>
   483→              <Label className="text-xs">Date & Time *</Label>
   484→              <Input type="datetime-local" value={form.date} onChange={e => setForm({ ...form, date: e.target.value })} className="mt-1" />
   485→            </div>
   486→          </div>
   487→          <div><Label className="text-xs">Main Dish *</Label><Input value={form.item} onChange={e => setForm({ ...form, item: e.target.value })} placeholder="e.g. Ugali & Sukuma Wiki" className="mt-1" /></div>
   488→          <div className="grid grid-cols-2 gap-3">
   489→            <div><Label className="text-xs">Accompaniment</Label><Input value={form.accompaniment} onChange={e => setForm({ ...form, accompaniment: e.target.value })} placeholder="e.g. Beef Stew" className="mt-1" /></div>
   490→            <div><Label className="text-xs">Beverage</Label><Input value={form.beverage} onChange={e => setForm({ ...form, beverage: e.target.value })} placeholder="e.g. Tea" className="mt-1" /></div>
   491→          </div>
   492→          <div className="grid grid-cols-2 gap-3">
   493→            <div><Label className="text-xs">Servings Planned</Label><Input type="number" value={form.servingsPlanned} onChange={e => setForm({ ...form, servingsPlanned: e.target.value })} className="mt-1" /></div>
   494→            <div><Label className="text-xs">Cook</Label><Input value={form.cook} onChange={e => setForm({ ...form, cook: e.target.value })} placeholder="Chef name" className="mt-1" /></div>
   495→          </div>
   496→          <div><Label className="text-xs">Notes</Label><Textarea value={form.notes} onChange={e => setForm({ ...form, notes: e.target.value })} className="mt-1" rows={2} /></div>
   497→          <div className="flex justify-end gap-2 pt-2">
   498→            <Button variant="outline" size="sm" onClick={onClose}>Cancel</Button>
   499→            <Button size="sm" onClick={handleSubmit} disabled={saving} className="bg-amber-600 hover:bg-amber-700">{saving ? 'Creating...' : 'Create Menu'}</Button>
   500→          </div>
   501→        </div>
   502→      </DialogContent>
   503→    </Dialog>
   504→  )
