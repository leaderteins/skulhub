    52→function bedStatusBadge(status: string): string {
    53→  if (status === 'Occupied') return 'border-amber-300 bg-amber-50 text-amber-700 dark:border-amber-800 dark:bg-amber-950 dark:text-amber-400'
    54→  if (status === 'Maintenance') return 'border-rose-300 bg-rose-50 text-rose-700 dark:border-rose-800 dark:bg-rose-950 dark:text-rose-400'
    55→  return 'border-emerald-300 bg-emerald-50 text-emerald-700 dark:border-emerald-800 dark:bg-emerald-950 dark:text-emerald-400'

    58→function roomStatusBadge(status: string): string {
    59→  if (status === 'Full') return 'border-rose-300 bg-rose-50 text-rose-700 dark:border-rose-800 dark:bg-rose-950 dark:text-rose-400'

    64→function scoreColor(score: number) {
    65→  if (score >= 8) return 'text-emerald-600 bg-emerald-50 border-emerald-200 dark:bg-emerald-950 dark:text-emerald-400 dark:border-emerald-800'

   309→function DormitoryDetailDialog({ dormId, onClose, onAssign }: { dormId: string; onClose: () => void; onAssign: () => void }) {
   310→  const { data: dorm, loading, refetch } = useFetch<any>(`/api/hostel/${dormId}`)
   311→  const [expandedRooms, setExpandedRooms] = useState<Record<string, boolean>>({})
   312→  const [vacateTarget, setVacateTarget] = useState<{ bedId: string; bedNumber: string; occupantName: string } | null>(null)
   313→  const [maintTarget, setMaintTarget] = useState<{ bedId: string; bedNumber: string; occupantName: string | null } | null>(null)
   314→  const [busyBedId, setBusyBedId] = useState<string | null>(null)
   315→  const [assignForRoom, setAssignForRoom] = useState<{ roomId: string; bedId: string | null } | null>(null)
   316→
   317→  const toggleRoom = (roomId: string) => setExpandedRooms(prev => ({ ...prev, [roomId]: !prev[roomId] }))
   318→
   319→  const handleVacate = async () => {
   320→    if (!vacateTarget) return
   321→    setBusyBedId(vacateTarget.bedId)
   322→    try {
   323→      await apiPost('/api/hostel/vacate', { bedId: vacateTarget.bedId, reason: 'Manual checkout' })
   324→      toast.success(`${vacateTarget.occupantName} checked out from ${vacateTarget.bedNumber}`)
   325→      setVacateTarget(null)
   326→      refetch()
   327→    } catch (e: any) {
   328→      toast.error(e?.message || 'Failed to vacate bed')
   329→    } finally {
   330→      setBusyBedId(null)
   331→    }
   332→  }
   333→
   334→  const handleMarkMaintenance = async (confirmVacate: boolean) => {
   335→    if (!maintTarget) return
   336→    setBusyBedId(maintTarget.bedId)
   337→    try {
   338→      const r = await fetch(`/api/hostel/beds/${maintTarget.bedId}`, {
   339→        method: 'PATCH',
   340→        headers: { 'Content-Type': 'application/json' },
   341→        body: JSON.stringify({ status: 'Maintenance', vacateIfOccupied: confirmVacate }),
   342→      })
   343→      const json = await r.json()
   344→      if (!r.ok) {
   345→        toast.error(json?.error || `Failed (${r.status})`)
   346→      } else {
   347→        toast.success(`${maintTarget.bedNumber} marked as Maintenance`)
   348→        setMaintTarget(null)
   349→        refetch()
   350→      }
   351→    } catch (e: any) {
   352→      toast.error(e?.message || 'Failed to update bed')
   353→    } finally {
   354→      setBusyBedId(null)
   355→    }
   356→  }
   357→
   358→  const handleRestore = async (bedId: string, bedNumber: string) => {
   359→    setBusyBedId(bedId)
   360→    try {
   361→      const r = await fetch(`/api/hostel/beds/${bedId}`, {
   362→        method: 'PATCH',
   363→        headers: { 'Content-Type': 'application/json' },
   364→        body: JSON.stringify({ status: 'Available' }),
   365→      })
   366→      const json = await r.json()
   367→      if (!r.ok) {
   368→        toast.error(json?.error || `Failed (${r.status})`)
   369→      } else {
   370→        toast.success(`${bedNumber} restored to Available`)
   371→        refetch()
   372→      }
   373→    } catch (e: any) {
   374→      toast.error(e?.message || 'Failed to restore bed')
   375→    } finally {
   376→      setBusyBedId(null)
   377→    }
   378→  }
   379→
   380→  return (
   381→    <>
   382→      <Dialog open onOpenChange={(o) => { if (!o) onClose() }}>
   383→        <DialogContent className="max-w-4xl max-h-[92vh] overflow-y-auto scrollbar-thin">
   384→          {loading || !dorm ? (
   385→            <div className="p-6 space-y-3">
   386→              <Skeleton className="h-12 w-full" />
   387→              <Skeleton className="h-32 w-full" />
   388→              <Skeleton className="h-48 w-full" />
   389→            </div>
   390→          ) : (
   391→            <>
   392→              <DialogHeader>
   393→                <DialogTitle className="flex items-center gap-2">
   394→                  <Home className="h-5 w-5 text-teal-500" /> {dorm.name}
   395→                </DialogTitle>
   396→                <DialogDescription>{dorm.gender} · {dorm.location} · {dorm.floors} floor{dorm.floors > 1 ? 's' : ''}</DialogDescription>
   397→              </DialogHeader>
   398→
   399→              {/* Info grid */}
   400→              <div className="grid gap-3 sm:grid-cols-2">
   401→                <Card className="border-teal-200/50 bg-teal-50/30 dark:bg-teal-950/10">
   402→                  <CardContent className="p-4">
   403→                    <p className="mb-2 text-[10px] font-semibold uppercase tracking-wider text-teal-700 dark:text-teal-400">Dormitory Info</p>
   404→                    <div className="space-y-1.5 text-sm">
   405→                      <div className="flex items-center gap-2"><Building2 className="h-3.5 w-3.5 text-muted-foreground" /><span>Capacity: {dorm.capacity} beds</span></div>
   406→                      <div className="flex items-center gap-2"><DoorOpen className="h-3.5 w-3.5 text-muted-foreground" /><span>Rooms: {dorm.rooms.length}</span></div>
   407→                      <div className="flex items-center gap-2"><MapPin className="h-3.5 w-3.5 text-muted-foreground" /><span>{dorm.location || 'N/A'}</span></div>
   408→                      <div className="flex items-center gap-2"><Badge variant="secondary" className={cn('text-[10px]', statusColor(dorm.status))}>{dorm.status}</Badge></div>
   409→                    </div>
   410→                  </CardContent>

   599→      {/* Per-bed assign dialog (uses same flow as top-level assign but with bed preselected) */}
   600→      {assignForRoom && (

   606→          dormitoryId={dorm?.id}
   607→          onAssigned={() => { setAssignForRoom(null); refetch() }}

   610→    </>

   706→interface AssignStudentDialogProps {
   707→  open: boolean
   708→  onOpenChange: (o: boolean) => void
   709→  preselectedBedId?: string | null
   710→  preselectedRoomId?: string | null
   711→  dormitoryId?: string
   712→  onAssigned?: () => void
   713→}
   714→
   715→interface StudentSearchResult {
   716→  id: string; admissionNo: string; firstName: string; lastName: string; gender: string
   717→  phone?: string | null; boarding: boolean; status: string; photoUrl?: string | null
   718→  currentEnrollment: any | null
   719→  currentBed: { id: string; bedNumber: string; room: { id: string; roomNumber: string; dormitory: { name: string; gender: string } } } | null
   720→}
   721→
   722→interface BedOption {
   723→  id: string; bedNumber: string; status: string; studentId: string | null
   724→  student: { id: string; admissionNo: string; firstName: string; lastName: string } | null
   725→  room: { id: string; roomNumber: string; capacity: number; occupied: number; status: string; dormitory: { id: string; name: string; gender: string } }
   726→}
   727→
   728→function AssignStudentDialog({ open, onOpenChange, preselectedBedId, preselectedRoomId, dormitoryId, onAssigned }: AssignStudentDialogProps) {
   729→  // 1) Fetch list of dormitories (with rooms) for the cascading selectors
   730→  const { data: hostelData, loading: loadingHostel } = useFetch<{ dormitories: any[] }>('/api/hostel')
   731→
   732→  // 2) Form state
   733→  const [dormId, setDormId] = useState<string | null>(dormitoryId || null)
   734→  const [roomId, setRoomId] = useState<string | null>(preselectedRoomId || null)
   735→  const [bedId, setBedId] = useState<string | null>(preselectedBedId || null)
   736→  const [studentQuery, setStudentQuery] = useState('')
   737→  const [selectedStudent, setSelectedStudent] = useState<StudentSearchResult | null>(null)
   738→  const [studentOpen, setStudentOpen] = useState(false)
   739→  const [submitting, setSubmitting] = useState(false)
   740→  const [warning, setWarning] = useState<{ code: string; message: string; details?: any } | null>(null)
   741→  const [notes, setNotes] = useState('')
   742→
   743→  // Reset state when dialog opens
   744→  useEffect(() => {
   745→    if (open) {
   746→      setDormId(dormitoryId || null)
   747→      setRoomId(preselectedRoomId || null)
   748→      setBedId(preselectedBedId || null)
   749→      setStudentQuery('')
   750→      setSelectedStudent(null)
   751→      setWarning(null)
   752→      setNotes('')
   753→    }
   754→  }, [open, preselectedBedId, preselectedRoomId, dormitoryId])
   755→
   756→  // Reset room/bed when dorm changes
   757→  useEffect(() => {
   758→    if (!open) return
   759→    if (dormId && dormId !== dormitoryId) {
   760→      setRoomId(null)
   761→      setBedId(null)
   762→    }
   763→  }, [dormId])
   764→
   765→  // Reset bed when room changes
   766→  useEffect(() => {
   767→    if (!open) return
   768→    if (roomId && roomId !== preselectedRoomId) {
   769→      setBedId(null)
   770→    }
   771→  }, [roomId])
   772→
   773→  // Filter dormitories/rooms/beds based on current selections
   774→  const dormitories = hostelData?.dormitories || []
   775→  const selectedDorm = dormitories.find((d: any) => d.id === dormId)
   776→  const rooms = useMemo(() => {
   777→    if (!dormId) return []
   778→    const d = dormitories.find((d: any) => d.id === dormId)
   779→    return d?.rooms || []
   780→  }, [dormId, dormitories])
   781→
   782→  // Fetch beds for the selected room (so we can show occupant info too)
   783→  const bedsUrl = roomId ? `/api/hostel/beds?roomId=${roomId}` : null
   784→  const { data: bedsData, loading: loadingBeds } = useFetch<{ beds: BedOption[] }>(bedsUrl, [roomId])
   785→  const beds = bedsData?.beds || []
   786→
   787→  // 3) Student search (debounced via useEffect on studentQuery)
   788→  const [studentResults, setStudentResults] = useState<StudentSearchResult[]>([])
   789→  const [loadingStudents, setLoadingStudents] = useState(false)
   790→  useEffect(() => {
   791→    if (!studentOpen) return
   792→    const q = studentQuery.trim()
   793→    if (q.length < 1) {
   794→      setStudentResults([])
   795→      return
   796→    }
   797→    let cancelled = false
   798→    setLoadingStudents(true)
   799→    const t = setTimeout(() => {
   800→      fetch(`/api/hostel/students/search?search=${encodeURIComponent(q)}&limit=20`)
   801→        .then(async (r) => {
   802→          if (!r.ok) return
   803→          const json = await r.json()
   804→          if (!cancelled) setStudentResults(json.students || [])
   805→        })
   806→        .catch(() => {})
   807→        .finally(() => { if (!cancelled) setLoadingStudents(false) })
   808→    }, 250)
   809→    return () => { cancelled = true; clearTimeout(t) }
   810→  }, [studentQuery, studentOpen])
   811→
   812→  const selectedBed = beds.find(b => b.id === bedId) || null
   813→
   814→  const canAssign = !!bedId && !!selectedStudent && !submitting
   815→
   816→  const handleAssign = async (force: boolean) => {
   817→    if (!bedId || !selectedStudent) return
   818→    setSubmitting(true)
   819→    setWarning(null)
   820→    try {
   821→      const r = await fetch('/api/hostel/assign', {
   822→        method: 'POST',
   823→        headers: { 'Content-Type': 'application/json' },
   824→        body: JSON.stringify({ bedId, studentId: selectedStudent.id, force, notes: notes.trim() || undefined }),
   825→      })
   826→      const json = await r.json()
   827→      if (!r.ok) {
   828→        if (r.status === 409 && (json.code === 'BED_OCCUPIED' || json.code === 'STUDENT_ALREADY_ASSIGNED' || json.code === 'GENDER_MISMATCH' || json.code === 'BED_MAINTENANCE' || json.code === 'ROOM_MAINTENANCE')) {
   829→          setWarning({
   830→            code: json.code,
   831→            message: json.error || 'Conflict',
   832→            details: json.currentOccupant || json.previousBed || null,
   833→          })
   834→          return
   835→        }
   836→        toast.error(json?.error || `Failed (${r.status})`)
   837→        return
   838→      }
   839→      if (json.noOp) {
   840→        toast.info(json.message || 'Student already assigned to this bed')
   841→      } else {
   842→        toast.success(`${selectedStudent.firstName} ${selectedStudent.lastName} assigned to ${selectedBed?.bedNumber || 'bed'}`)
   843→      }
   844→      onOpenChange(false)
   845→      onAssigned?.()
   846→    } catch (e: any) {
   847→      toast.error(e?.message || 'Failed to assign student')
   848→    } finally {
   849→      setSubmitting(false)
   850→    }
   851→  }
   852→
   853→  return (
   854→    <Dialog open={open} onOpenChange={(o) => { if (!submitting) onOpenChange(o) }}>
   855→      <DialogContent className="max-w-2xl max-h-[92vh] overflow-y-auto scrollbar-thin">
   856→        <DialogHeader>
   857→          <DialogTitle className="flex items-center gap-2">
   858→            <UserPlus className="h-5 w-5 text-teal-500" /> Assign Student to Bed
   859→          </DialogTitle>
   860→          <DialogDescription>
   861→            Select a hostel, room, and bed, then search for a student by name or admission number.
   862→          </DialogDescription>
   863→        </DialogHeader>
   864→
   865→        <div className="space-y-4">
   866→          {/* Step 1: Hostel */}
   867→          <div className="space-y-1.5">
   868→            <Label className="text-xs font-medium text-muted-foreground">1. Select Hostel</Label>
   869→            {loadingHostel ? (
   870→              <Skeleton className="h-10 w-full" />
   871→            ) : (
   872→              <Select value={dormId || ''} onValueChange={setDormId}>
   873→                <SelectTrigger><SelectValue placeholder="Choose a dormitory..." /></SelectTrigger>
   874→                <SelectContent>
   875→                  {dormitories.map((d: any) => (
