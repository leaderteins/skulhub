   650→function GradeEntryDialog({ assessment, onClose }: { assessment: any; onClose: () => void }) {
   651→  // Fetch students enrolled in this assessment's class level
   652→  const { data: studentsData, loading } = useFetch<any>(
   653→    `/api/students?classLevel=${assessment.classLevelId || ''}&pageSize=200`
   654→  )
   655→  const [grades, setGrades] = useState<Record<string, string>>({})
   656→  const [comments, setComments] = useState<Record<string, string>>({})
   657→  const [selectedStudent, setSelectedStudent] = useState<string | null>(null)
   658→  const [saving, setSaving] = useState(false)
   659→
   660→  const students = studentsData?.students || []
   661→
   662→  const handleSave = async () => {
   663→    const entries = Object.entries(grades).filter(([, m]) => m !== '' && m != null)
   664→    if (entries.length === 0) {
   665→      toast.error('Enter at least one mark')
   666→      return
   667→    }
   668→    setSaving(true)
   669→    try {
   670→      let saved = 0
   671→      for (const [studentId, marks] of entries) {
   672→        const numMarks = Number(marks)
   673→        if (!Number.isFinite(numMarks) || numMarks < 0 || numMarks > assessment.totalMarks) continue
   674→        try {
   675→          await apiPost('/api/grades', {
   676→            studentId,
   677→            subjectId: assessment.subjectId,
   678→            examId: assessment.id,
   679→            marks: numMarks,
   680→            totalMarks: assessment.totalMarks,
   681→            source: 'internal',
   682→            teacherComment: comments[studentId]?.trim() || null,
   683→            remarks: comments[studentId]?.trim() || null,
   684→          })
   685→          saved++
   686→        } catch { /* skip individual errors */ }
   687→      }
   688→      toast.success(`${saved} grade${saved !== 1 ? 's' : ''} saved`, {
   689→        description: `${assessment.title} · ${assessment.subject?.name}`,
   690→      })
   691→      onClose()
   692→    } catch (e: any) {
   693→      toast.error(e?.message || 'Failed to save grades')
   694→    } finally {
   695→      setSaving(false)
   696→    }
   697→  }
   698→
   699→  return (
   700→    <Dialog open onOpenChange={(o) => { if (!o) onClose() }}>
   701→      <DialogContent className="max-w-3xl max-h-[92vh] overflow-y-auto scrollbar-thin">
   702→        <DialogHeader>
   703→          <DialogTitle className="flex items-center gap-2">
   704→            <ClipboardEdit className="h-5 w-5 text-teal-500" /> Record Grades & Remarks
   705→          </DialogTitle>
   706→          <DialogDescription>
   707→            {assessment.title} · {assessment.subject?.name} · Max {assessment.totalMarks} marks
   708→          </DialogDescription>
   709→        </DialogHeader>
   710→        {loading ? (
   711→          <Skeleton className="h-48 w-full" />
   712→        ) : students.length === 0 ? (
   713→          <div className="py-8 text-center text-sm text-muted-foreground">
   714→            No students found in this class level.
   715→          </div>
   716→        ) : (
   717→          <div className="space-y-2">
   718→            <Table>
   719→              <TableHeader>
   720→                <TableRow>
   721→                  <TableHead className="text-xs">Adm No</TableHead>
   722→                  <TableHead className="text-xs">Student Name</TableHead>
   723→                  <TableHead className="text-xs text-right">Marks / {assessment.totalMarks}</TableHead>
   724→                  <TableHead className="text-xs">CBE AL</TableHead>
   725→                  <TableHead className="text-xs">Descriptor</TableHead>
   726→                  <TableHead className="text-xs">Teacher Comment</TableHead>
   727→                </TableRow>
   728→              </TableHeader>
   729→              <TableBody>
   730→                {students.map((s: any) => {
   731→                  const marks = grades[s.id]
   732→                  const pct = marks ? (Number(marks) / assessment.totalMarks) * 100 : 0
   733→                  // Official CBE 8-Level Achievement Level scale
   734→                  let alLevel = '—'
   735→                  let descriptor = '—'
   736→                  let badgeColor = 'border-slate-300 text-slate-600'
   737→                  if (marks) {
   738→                    if (pct >= 90) { alLevel = 'AL 8'; descriptor = 'EE1'; badgeColor = 'border-emerald-400 text-emerald-700 bg-emerald-50' }
   739→                    else if (pct >= 75) { alLevel = 'AL 7'; descriptor = 'EE2'; badgeColor = 'border-emerald-300 text-emerald-700 bg-emerald-50' }
   740→                    else if (pct >= 58) { alLevel = 'AL 6'; descriptor = 'ME1'; badgeColor = 'border-teal-300 text-teal-700 bg-teal-50' }
   741→                    else if (pct >= 41) { alLevel = 'AL 5'; descriptor = 'ME2'; badgeColor = 'border-cyan-300 text-cyan-700 bg-cyan-50' }
   742→                    else if (pct >= 31) { alLevel = 'AL 4'; descriptor = 'AE1'; badgeColor = 'border-amber-300 text-amber-700 bg-amber-50' }
   743→                    else if (pct >= 21) { alLevel = 'AL 3'; descriptor = 'AE2'; badgeColor = 'border-orange-300 text-orange-700 bg-orange-50' }
   744→                    else if (pct >= 11) { alLevel = 'AL 2'; descriptor = 'BE1'; badgeColor = 'border-rose-300 text-rose-700 bg-rose-50' }
   745→                    else { alLevel = 'AL 1'; descriptor = 'BE2'; badgeColor = 'border-rose-500 text-rose-800 bg-rose-100' }
   746→                  }
   747→                  return (

   820→// ExternalMarksDialog — add marks from external exams (KCPE, KCSE, mock, etc.)
   821→// Allows choosing which subject the external marks belong to.
   822→// ---------------------------------------------------------------------------
   823→function ExternalMarksDialog({ assessment, onClose }: { assessment: any; onClose: () => void }) {
   824→  const [externalSource, setExternalSource] = useState('KCSE Mock')
   825→  const [subjectId, setSubjectId] = useState(assessment.subjectId || '')
   826→  const [classLevelId, setClassLevelId] = useState(assessment.classLevelId || '')
   827→  const [examName, setExamName] = useState('')
   828→  const [totalMarks, setTotalMarks] = useState('100')
   829→  const [marksText, setMarksText] = useState('')
   830→  const [saving, setSaving] = useState(false)
   831→
   832→  const { data: studentsData, loading } = useFetch<any>(
   833→    classLevelId ? `/api/students?classLevel=${classLevelId}&pageSize=200` : null
   834→  )
   835→  const { data: subjectsData } = useFetch<any>('/api/academics/subjects')
   836→  const { data: classLevelsData } = useFetch<any>('/api/academics/class-levels')
   837→  const students = studentsData?.students || []
   838→
   839→  // Parse the pasted text (format: "admNo,marks" per line, or "name,marks")
   840→  const parsedRows = marksText
   841→    .split('\n')
   842→    .map(line => line.trim())
   843→    .filter(Boolean)
   844→    .map(line => {
   845→      const parts = line.split(/[,\t]/).map(p => p.trim())
   846→      return { admNoOrName: parts[0], marks: parts[1] || '' }
   847→    })
   848→
   849→  const handleSave = async () => {
   850→    if (parsedRows.length === 0) {
   851→      toast.error('Paste at least one row of marks (format: AdmissionNo,Marks)')
   852→      return
   853→    }
   854→    if (!subjectId) {
   855→      toast.error('Select the subject these marks belong to')
   856→      return
   857→    }
   858→    setSaving(true)
   859→    try {
   860→      let saved = 0
   861→      let skipped = 0
   862→      for (const row of parsedRows) {
   863→        // Match by admission number or name
   864→        const student = students.find((s: any) =>
   865→          s.admissionNo?.toLowerCase() === row.admNoOrName.toLowerCase() ||
   866→          `${s.firstName} ${s.lastName}`.toLowerCase().includes(row.admNoOrName.toLowerCase())
   867→        )
   868→        if (!student) { skipped++; continue }
   869→        const numMarks = Number(row.marks)
   870→        if (!Number.isFinite(numMarks) || numMarks < 0) { skipped++; continue }
   871→        try {
   872→          await apiPost('/api/grades', {
   873→            studentId: student.id,
   874→            subjectId,
   875→            examId: assessment.id,
   876→            marks: numMarks,
   877→            totalMarks: Number(totalMarks),
   878→            source: 'external',
   879→            externalSource,
   880→            remarks: `External: ${externalSource}`,
   881→          })
   882→          saved++
   883→        } catch { skipped++ }
   884→      }
   885→      toast.success(`${saved} external marks imported`, {
   886→        description: skipped > 0 ? `${skipped} row(s) skipped (student not found or invalid marks)` : undefined,
   887→      })
   888→      if (saved > 0) onClose()
   889→    } catch (e: any) {
   890→      toast.error(e?.message || 'Failed to import external marks')
   891→    } finally {
   892→      setSaving(false)
   893→    }
   894→  }
   895→
   896→  return (
   897→    <Dialog open onOpenChange={(o) => { if (!o) onClose() }}>
   898→      <DialogContent className="max-w-2xl max-h-[92vh] overflow-y-auto scrollbar-thin">
   899→        <DialogHeader>
   900→          <DialogTitle className="flex items-center gap-2">
   901→            <FilePlus className="h-5 w-5 text-amber-500" /> Add External Marks
   902→          </DialogTitle>
   903→          <DialogDescription>
   904→            Import marks from external exams (KCPE, KCSE, Mock, etc.) not assessed in the system.
   905→          </DialogDescription>
   906→        </DialogHeader>
   907→        <div className="space-y-3">
   908→          <div className="rounded-lg border border-amber-200 bg-amber-50/50 p-3 dark:border-amber-800 dark:bg-amber-950/20">
   909→            <p className="text-xs text-amber-800 dark:text-amber-300">
   910→              <strong>How it works:</strong> Choose the subject and class level these marks belong to,
   911→              then paste a list of <code className="rounded bg-amber-100 px-1">AdmissionNumber,Marks</code>
   912→              (one per line). The system matches each row to a student and saves the grade.
   913→            </p>
   914→          </div>
   915→
   916→          {/* Subject selector — lets you choose which subject to add marks to */}
   917→          <div>
   918→            <Label className="text-xs">Subject for these marks *</Label>
   919→            <Select value={subjectId} onValueChange={setSubjectId}>
   920→              <SelectTrigger className="mt-1"><SelectValue placeholder="Select subject" /></SelectTrigger>
   921→              <SelectContent>
   922→                {(subjectsData?.subjects || subjectsData || []).map((s: any) => (
   923→                  <SelectItem key={s.id} value={s.id}>{s.name} ({s.code})</SelectItem>
   924→                ))}
   925→              </SelectContent>
   926→            </Select>
   927→            <p className="mt-1 text-[10px] text-muted-foreground">
   928→              You can choose a different subject than the assessment's default subject.
   929→            </p>
   930→          </div>
   931→
   932→          <div className="grid grid-cols-2 gap-3">
   933→            <div>
   934→              <Label className="text-xs">Class Level</Label>
   935→              <Select value={classLevelId} onValueChange={setClassLevelId}>
   936→                <SelectTrigger className="mt-1"><SelectValue placeholder="Select class" /></SelectTrigger>
   937→                <SelectContent>
   938→                  {(classLevelsData?.classLevels || classLevelsData || []).map((c: any) => (
   939→                    <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   940→                  ))}
   941→                </SelectContent>
   942→              </Select>
   943→            </div>
   944→            <div>
   945→              <Label className="text-xs">External Source</Label>
   946→              <Select value={externalSource} onValueChange={setExternalSource}>
   947→                <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   948→                <SelectContent>
   949→                  <SelectItem value="KCSE Mock">KCSE Mock</SelectItem>
   950→                  <SelectItem value="KCSE">KCSE (National)</SelectItem>
   951→                  <SelectItem value="KCPE">KCPE (National)</SelectItem>
   952→                  <SelectItem value="Joint Mock">Joint Mock (Multiple Schools)</SelectItem>
   953→                  <SelectItem value="Opener Exam">Opener Exam (External)</SelectItem>
   954→                  <SelectItem value="Other">Other</SelectItem>
   955→                </SelectContent>
   956→              </Select>
   957→            </div>
   958→          </div>
   959→
   960→          <div className="grid grid-cols-2 gap-3">
   961→            <div>
   962→              <Label className="text-xs">Exam Name (optional)</Label>
   963→              <Input value={examName} onChange={e => setExamName(e.target.value)} placeholder="e.g. Term 2 Joint Mock" className="mt-1" />
   964→            </div>
   965→            <div>
   966→              <Label className="text-xs">Total Marks</Label>
   967→              <Input type="number" value={totalMarks} onChange={e => setTotalMarks(e.target.value)} className="mt-1" />
   968→            </div>
   969→          </div>
   970→
   971→          <div>
   972→            <Label className="text-xs">Paste marks (one per line: AdmissionNo,Marks)</Label>
   973→            <Textarea
   974→              value={marksText}
   975→              onChange={e => setMarksText(e.target.value)}
   976→              placeholder={'ADM/001,85\nADM/002,72\nADM/003,91'}
   977→              className="mt-1 font-mono text-xs"
   978→              rows={6}
   979→            />
   980→            <p className="mt-1 text-[10px] text-muted-foreground">
   981→              {parsedRows.length} row(s) detected · {students.length} students in class
   982→            </p>
   983→          </div>
   984→
   985→          <div className="flex justify-end gap-2 pt-2">
   986→            <Button variant="outline" size="sm" onClick={onClose}>Cancel</Button>
   987→            <Button size="sm" onClick={handleSave} disabled={saving} className="bg-amber-600 hover:bg-amber-700">
   988→              {saving ? <Loader2 className="h-4 w-4 animate-spin" /> : <Upload className="h-4 w-4" />}
   989→              Import {parsedRows.length > 0 ? `(${parsedRows.length})` : ''}
   990→            </Button>
   991→          </div>
   992→        </div>
   993→      </DialogContent>
   994→    </Dialog>
   995→  )
   996→}

   999→// ChangeSubjectDialog — change the subject assigned to an assessment
  1000→// (in case it was wrongly chosen when creating)
  1001→// ---------------------------------------------------------------------------
  1002→function ChangeSubjectDialog({ assessment, onClose }: { assessment: any; onClose: () => void }) {
  1003→  const [subjectId, setSubjectId] = useState(assessment.subjectId || '')
  1004→  const [saving, setSaving] = useState(false)
  1005→  const { data: subjectsData } = useFetch<any>('/api/academics/subjects')
  1006→  const subjects = subjectsData?.subjects || subjectsData || []
  1007→
  1008→  const handleSave = async () => {
  1009→    if (!subjectId) { toast.error('Select a subject'); return }
  1010→    if (subjectId === assessment.subjectId) {
  1011→      toast.info('Subject unchanged')
  1012→      onClose()
  1013→      return
  1014→    }
  1015→    setSaving(true)
  1016→    try {
  1017→      await apiPut(`/api/exams/${assessment.id}`, { subjectId, entityType: 'assessment' })
  1018→      toast.success('Subject updated', {
  1019→        description: `Changed from ${assessment.subject?.name} to ${subjects.find((s: any) => s.id === subjectId)?.name}`,
  1020→      })
  1021→      onClose()
  1022→      // Trigger a refetch of the assessment
  1023→      setTimeout(() => window.location.reload(), 500)
  1024→    } catch (e: any) {
  1025→      toast.error(e?.message || 'Failed to change subject')
  1026→    } finally {
  1027→      setSaving(false)
  1028→    }
  1029→  }
  1030→
  1031→  return (
  1032→    <Dialog open onOpenChange={(o) => { if (!o) onClose() }}>
  1033→      <DialogContent className="max-w-md">
  1034→        <DialogHeader>
  1035→          <DialogTitle className="flex items-center gap-2">
  1036→            <Edit3 className="h-5 w-5 text-teal-500" /> Change Subject
  1037→          </DialogTitle>
  1038→          <DialogDescription>
  1039→            Current: {assessment.subject?.name} — select a new subject for this assessment.
  1040→          </DialogDescription>
  1041→        </DialogHeader>
  1042→        <div className="space-y-3">
  1043→          <div>
  1044→            <Label className="text-xs">New Subject</Label>
  1045→            <Select value={subjectId} onValueChange={setSubjectId}>
  1046→              <SelectTrigger className="mt-1"><SelectValue placeholder="Select new subject" /></SelectTrigger>
  1047→              <SelectContent>
  1048→                {subjects.map((s: any) => (
  1049→                  <SelectItem key={s.id} value={s.id}>
  1050→                    {s.name} ({s.code})
  1051→                    {s.id === assessment.subjectId && ' (current)'}
  1052→                  </SelectItem>
  1053→                ))}
  1054→              </SelectContent>
  1055→            </Select>
  1056→          </div>
  1057→          <div className="rounded-lg border border-amber-200 bg-amber-50/50 p-3 dark:border-amber-800 dark:bg-amber-950/20">
  1058→            <p className="text-[10px] text-amber-800 dark:text-amber-300">
  1059→              <strong>Note:</strong> Changing the subject will move all grades recorded for this
  1060→              assessment to the new subject. This cannot be undone.
  1061→            </p>
  1062→          </div>
  1063→          <div className="flex justify-end gap-2 pt-2">
  1064→            <Button variant="outline" size="sm" onClick={onClose}>Cancel</Button>
  1065→            <Button size="sm" onClick={handleSave} disabled={saving} className="bg-teal-600 hover:bg-teal-700">
  1066→              {saving ? <Loader2 className="h-4 w-4 animate-spin" /> : <Save className="h-4 w-4" />}
  1067→              Change Subject
  1068→            </Button>
  1069→          </div>
  1070→        </div>
  1071→      </DialogContent>
  1072→    </Dialog>
  1073→  )
