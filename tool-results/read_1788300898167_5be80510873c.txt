   133→function StatusBadge({ status }: { status: string }) {
   134→  const s = status.toLowerCase()
   135→  const styles: Record<string, string> = {
   136→    active: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300',
   137→    trial: 'bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300',
   138→    suspended: 'bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-300',
   139→    expired: 'bg-slate-100 text-slate-600 dark:bg-slate-800 dark:text-slate-400',
   140→  }
   141→  return (
   142→    <span className={cn('inline-flex items-center gap-1 rounded-full px-2.5 py-0.5 text-xs font-semibold', styles[s] || styles.expired)}>
   143→      <span className={cn('h-1.5 w-1.5 rounded-full', s === 'active' ? 'bg-emerald-500' : s === 'trial' ? 'bg-amber-500' : s === 'suspended' ? 'bg-rose-500' : 'bg-slate-400')} />
   144→      {status}
   145→    </span>
   146→  )
   147→}
   148→
   149→function PlanBadge({ plan }: { plan: string }) {
   150→  return (
   151→    <span className={cn('inline-flex items-center gap-1 rounded-md px-2 py-0.5 text-xs font-semibold', PLAN_COLOR[plan] || PLAN_COLOR.Starter)}>
   152→      {plan === 'Enterprise' && <Crown className="h-3 w-3" />}
   153→      {plan === 'Premium' && <Sparkles className="h-3 w-3" />}
   154→      {plan}
   155→    </span>
   156→  )
   157→}
