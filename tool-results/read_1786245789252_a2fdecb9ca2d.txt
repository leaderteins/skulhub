   504→            </div>
   505→          </>

   507→      </DialogContent>

   515→function AddEventDialog({ onClose }: { onClose: () => void }) {
   516→  const [form, setForm] = useState({
   517→    title: '', description: '', category: 'General', startDate: '', endDate: '',
   518→    allDay: false, location: '', organizer: '', audience: 'All', priority: 'Normal', color: 'emerald',
   519→  })
