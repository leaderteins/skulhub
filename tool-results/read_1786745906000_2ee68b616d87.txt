   817→function initialsOf(name: string): string {
   818→  const parts = name.trim().split(/\s+/)
   819→  return ((parts[0]?.[0] || '') + (parts[1]?.[0] || '')).toUpperCase()
   820→}
   821→
   822→function roleColor(role: string): string {
   823→  const r = role.toLowerCase()
   824→  if (r.includes('principal')) return 'bg-emerald-600'
   825→  if (r.includes('bursar')) return 'bg-teal-600'
   826→  if (r.includes('teacher') || r.includes('hod')) return 'bg-cyan-600'
   827→  if (r.includes('librar')) return 'bg-amber-600'
   828→  if (r.includes('clerk')) return 'bg-violet-600'
   829→  return 'bg-slate-500'
   830→}
   831→
   832→function roleBadgeColor(role: string): string {
   833→  const r = role.toLowerCase()
   834→  if (r.includes('principal')) return 'bg-emerald-50 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-400'
   835→  if (r.includes('bursar')) return 'bg-teal-50 text-teal-700 dark:bg-teal-950 dark:text-teal-400'
   836→  if (r.includes('teacher') || r.includes('hod')) return 'bg-cyan-50 text-cyan-700 dark:bg-cyan-950 dark:text-cyan-400'
   837→  if (r.includes('librar')) return 'bg-amber-50 text-amber-700 dark:bg-amber-950 dark:text-amber-400'
   838→  if (r.includes('clerk')) return 'bg-violet-50 text-violet-700 dark:bg-violet-950 dark:text-violet-400'
   839→  return 'bg-slate-50 text-slate-700 dark:bg-slate-800 dark:text-slate-400'
   840→}
   841→
   842→function statusBadgeColor(status: string): string {
   843→  const s = status.toLowerCase()
   844→  if (s === 'active') return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-400'
   845→  if (s === 'suspended') return 'bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-400'
   846→  if (s === 'invited') return 'bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-400'
   847→  return 'bg-slate-100 text-slate-700 dark:bg-slate-800 dark:text-slate-400'
   848→}
