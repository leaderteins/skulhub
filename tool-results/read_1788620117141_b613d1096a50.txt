   242→// ---------------------------------------------------------------------------
   243→// Derived metrics — computed client-side from the schools list
   244→// (Honest proxies since the API only exposes monthlyGrowth & per-school revenue)

   246→function buildMonthlyRevenue(schools: SchoolListItem[], months: Array<{ label: string; key: string }>) {
   247→  const buckets: Record<string, number> = {}
   248→  for (const m of months) buckets[m.key] = 0
   249→  for (const s of schools) {
   250→    const d = new Date(s.createdAt)
   251→    const key = `${d.getFullYear()}-${d.getMonth()}`
   252→    if (buckets[key] !== undefined) buckets[key] += s.revenue || 0
   253→  }
   254→  return months.map(m => ({ label: m.label, amount: buckets[m.key] || 0 }))
   255→}
   256→
   257→function buildMonthlyStudents(schools: SchoolListItem[], months: Array<{ label: string; key: string }>) {
   258→  const buckets: Record<string, number> = {}
   259→  for (const m of months) buckets[m.key] = 0
   260→  for (const s of schools) {
   261→    const d = new Date(s.createdAt)
   262→    const key = `${d.getFullYear()}-${d.getMonth()}`
   263→    if (buckets[key] !== undefined) buckets[key] += s.studentCount || 0
   264→  }
   265→  return months.map(m => ({ label: m.label, count: buckets[m.key] || 0 }))
   266→}
   267→
   268→function buildMonthlyTrials(schools: SchoolListItem[], months: Array<{ label: string; key: string }>) {
   269→  const buckets: Record<string, number> = {}
   270→  for (const m of months) buckets[m.key] = 0
   271→  for (const s of schools) {
   272→    if (!s.trialEndsAt) continue
   273→    const d = new Date(s.trialEndsAt)
   274→    const key = `${d.getFullYear()}-${d.getMonth()}`
   275→    if (buckets[key] !== undefined) buckets[key] += 1
   276→  }
   277→  return months.map(m => ({ label: m.label, count: buckets[m.key] || 0 }))
   278→}
   279→
   280→function buildMonthKeys(monthlyGrowth: Array<{ label: string; count: number }>): Array<{ label: string; key: string }> {
   281→  // Reconstruct month keys from current date backwards, matching API labels
   282→  const now = new Date()
   283→  const keys: Array<{ label: string; key: string }> = []
   284→  for (let i = 5; i >= 0; i--) {
   285→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1)
   286→    const label = d.toLocaleDateString('en-KE', { month: 'short' })
   287→    const key = `${d.getFullYear()}-${d.getMonth()}`
   288→    keys.push({ label, key })
   289→  }
   290→  // If API gave us labels, use them (prefers server-derived labels)
   291→  if (monthlyGrowth.length === keys.length) {
   292→    return monthlyGrowth.map((m, i) => ({ label: m.label, key: keys[i].key }))
   293→  }
   294→  return keys
   295→}

   297→function pctDelta(series: number[]): number | null {
   298→  if (series.length < 2) return null
   299→  const last = series[series.length - 1]
   300→  const prev = series[series.length - 2]
   301→  if (prev === 0) return last > 0 ? 100 : 0
   302→  return ((last - prev) / prev) * 100
   303→}
